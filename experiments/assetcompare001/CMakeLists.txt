cmake_minimum_required(VERSION 3.24)
project(WILDSIGNAL_ASSETCOMPARE001 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(CTest)
include(FetchContent)
add_subdirectory(../hearthgrove001 inherited_hearthgrove)
if(PIXEL3D_BUILD_GL)
 add_executable(wildsignal_assetcompare001 src/main.cpp src/gpu.cpp src/hud.cpp src/image.cpp src/mesh_gpu.cpp)
 target_include_directories(wildsignal_assetcompare001 BEFORE PRIVATE src ../hearthgrove001/src)
 FetchContent_GetProperties(stb SOURCE_DIR COMPARE_STB)
 target_include_directories(wildsignal_assetcompare001 PRIVATE ${COMPARE_STB})
 target_link_libraries(wildsignal_assetcompare001 PRIVATE hearthgrove_core SDL3::SDL3-static)
 if(WIN32)
  target_link_libraries(wildsignal_assetcompare001 PRIVATE psapi)
 endif()
 execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} OUTPUT_VARIABLE SOURCE_ID OUTPUT_STRIP_TRAILING_WHITESPACE)
 target_compile_definitions(wildsignal_assetcompare001 PRIVATE PIXEL3D_SOURCE_ID="${SOURCE_ID}")
 install(TARGETS wildsignal_assetcompare001 RUNTIME DESTINATION . COMPONENT AssetCompare)
 install(DIRECTORY launchers/ DESTINATION . COMPONENT AssetCompare)
 install(FILES README.md DESTINATION . COMPONENT AssetCompare)
 FetchContent_GetProperties(sdl3 SOURCE_DIR COMPARE_SDL)
 FetchContent_GetProperties(cgltf SOURCE_DIR COMPARE_CGLTF)
 install(FILES ${COMPARE_SDL}/LICENSE.txt DESTINATION licenses RENAME SDL3.txt COMPONENT AssetCompare)
 install(FILES ${COMPARE_CGLTF}/LICENSE DESTINATION licenses RENAME cgltf.txt COMPONENT AssetCompare)
 install(FILES ${COMPARE_STB}/LICENSE DESTINATION licenses RENAME stb.txt COMPONENT AssetCompare)
endif()
if(BUILD_TESTING)
 add_executable(assetcompare_navigation_tests tests/navigation.cpp)
 target_include_directories(assetcompare_navigation_tests PRIVATE src)
 target_link_libraries(assetcompare_navigation_tests PRIVATE hearthgrove_core)
 add_test(NAME assetcompare001_camera_movement_collision COMMAND assetcompare_navigation_tests)
endif()
