cmake_minimum_required(VERSION 3.24)
project(WILDSIGNAL_PETALYN_REVIEW LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
# Isolated opt-in executable; no private asset bytes or automatic fixture download.
include(CTest)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_subdirectory(../creature3d003 inherited_creature3d003)
target_compile_definitions(pixel3d_core PRIVATE PETALYN_RUNTIME=1)
if(TARGET wildsignal_creature3d003)
 target_compile_definitions(wildsignal_creature3d003 PRIVATE PETALYN_RUNTIME=1)
 if(MSVC)
  target_compile_options(wildsignal_creature3d003 PRIVATE /utf-8)
 endif()
 set_target_properties(wildsignal_creature3d003 PROPERTIES OUTPUT_NAME wildsignal_petalyn)
endif()
set(PETALYN_ASSETS "" CACHE PATH "Owner's private Petalyn runtime directory, never in public Git")
if(BUILD_TESTING)
 add_executable(petalyn_animation_tests tests/animation.cpp)
 target_link_libraries(petalyn_animation_tests PRIVATE pixel3d_core)
 add_test(NAME petalyn_skin_math COMMAND petalyn_animation_tests)
 if(PETALYN_ASSETS)
  foreach(lod RANGE 0 2)
   add_test(NAME petalyn_lod${lod}_skin_and_clips COMMAND petalyn_animation_tests ${PETALYN_ASSETS}/WS010_LOD${lod}.glb)
  endforeach()
  if(PIXEL3D_BUILD_GL AND PIXEL3D_TEST_GL)
   add_test(NAME petalyn_actual_gpu_playback COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/runtime.py $<TARGET_FILE:wildsignal_creature3d003> ${PETALYN_ASSETS})
  endif()
 endif()
endif()
