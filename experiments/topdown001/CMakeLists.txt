cmake_minimum_required(VERSION 3.24)
project(WILDSIGNAL_TOPDOWN001 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(WIN32 AND NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 only")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(TOPDOWN001_BUILD_SDL "Build interactive SDL3 prototype" ON)
option(TOPDOWN001_BUILD_LEGACY "Build preserved libpng proof when PNG is available" ON)
option(TOPDOWN001_SANITIZE "Enable address/undefined behavior sanitizers" OFF)
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} OUTPUT_VARIABLE TOPDOWN_BUILD_ID OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
execute_process(COMMAND git status --porcelain WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} OUTPUT_VARIABLE TD_DIRTY OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NOT TOPDOWN_BUILD_ID)
  set(TOPDOWN_BUILD_ID "UNKNOWN_SOURCE")
endif()
if(TD_DIRTY)
  string(APPEND TOPDOWN_BUILD_ID "-dirty")
endif()
add_library(topdown001_core src/playable.cpp ../../src/engine001/json.cpp)
target_include_directories(topdown001_core PUBLIC src ../../src/engine001)
target_compile_definitions(topdown001_core PUBLIC TOPDOWN_BUILD_ID="${TOPDOWN_BUILD_ID}")
if(MSVC)
  target_compile_options(topdown001_core PUBLIC /W4 /permissive- /utf-8)
else()
  target_compile_options(topdown001_core PUBLIC -Wall -Wextra -Wpedantic -Werror)
  if(TOPDOWN001_SANITIZE)
    target_compile_options(topdown001_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(topdown001_core PUBLIC -fsanitize=address,undefined)
  endif()
endif()
if(WIN32)
  target_compile_definitions(topdown001_core PUBLIC _WIN32_WINNT=0x0601 WIN32_LEAN_AND_MEAN NOMINMAX)
endif()
add_executable(wildsignal_topdown001_headless src/runner.cpp)
target_link_libraries(wildsignal_topdown001_headless PRIVATE topdown001_core)
if(WIN32)
  target_link_libraries(wildsignal_topdown001_headless PRIVATE psapi)
endif()
if(TOPDOWN001_BUILD_SDL)
  include(FetchContent)
  set(SDL_SHARED OFF CACHE BOOL "" FORCE)
  set(SDL_STATIC ON CACHE BOOL "" FORCE)
  set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
  set(SDL_TESTS OFF CACHE BOOL "" FORCE)
  set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
  foreach(subsystem AUDIO CAMERA GPU HAPTIC POWER SENSOR DIALOG)
    set(SDL_${subsystem} OFF CACHE BOOL "" FORCE)
  endforeach()
  set(SDL_RENDER_D3D12 OFF CACHE BOOL "" FORCE)
  set(SDL_RENDER_GPU OFF CACHE BOOL "" FORCE)
  set(SDL_VULKAN OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(SDL3 GIT_REPOSITORY https://github.com/libsdl-org/SDL.git GIT_TAG 877399b2b2cf21e67554ed9046410f268ce1d1b2 GIT_SHALLOW FALSE)
  FetchContent_MakeAvailable(SDL3)
  add_executable(wildsignal_topdown001_playable src/runner.cpp)
  target_compile_definitions(wildsignal_topdown001_playable PRIVATE TOPDOWN_WITH_SDL=1)
  target_link_libraries(wildsignal_topdown001_playable PRIVATE topdown001_core SDL3::SDL3-static)
  if(WIN32)
    target_link_libraries(wildsignal_topdown001_playable PRIVATE psapi)
  endif()
  install(TARGETS wildsignal_topdown001_playable wildsignal_topdown001_headless RUNTIME DESTINATION .)
  install(FILES ${sdl3_SOURCE_DIR}/LICENSE.txt DESTINATION licenses RENAME SDL3.txt)
  install(FILES README.playable.md DESTINATION .)
  install(DIRECTORY data/ DESTINATION data)
  install(FILES tools/import_assets.py DESTINATION tools)
endif()
if(TOPDOWN001_BUILD_LEGACY)
  find_package(PNG QUIET)
  if(PNG_FOUND)
    add_executable(wildsignal_topdown001 src/main.cpp)
    target_link_libraries(wildsignal_topdown001 PRIVATE PNG::PNG)
  else()
    message(STATUS "Preserved legacy proof NOT BUILT: supply libpng development files")
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(topdown001_tests tests/core_tests.cpp)
  target_link_libraries(topdown001_tests PRIVATE topdown001_core)
  add_test(NAME topdown001_invariants COMMAND topdown001_tests)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME topdown001_synthetic_boot COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:wildsignal_topdown001_headless> ${CMAKE_CURRENT_SOURCE_DIR})
  if(TOPDOWN001_BUILD_SDL)
    add_test(NAME topdown001_sdl_input_resize COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:wildsignal_topdown001_headless> ${CMAKE_CURRENT_SOURCE_DIR} --native $<TARGET_FILE:wildsignal_topdown001_playable>)
  endif()
endif()
