cmake_minimum_required(VERSION 3.16)
project(WILDSIGNAL_TOPDOWN001 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(PNG REQUIRED)
add_executable(wildsignal_topdown001 src/main.cpp)
target_link_libraries(wildsignal_topdown001 PRIVATE PNG::PNG)
if(MSVC)
  target_compile_options(wildsignal_topdown001 PRIVATE /W4 /permissive-)
else()
  target_compile_options(wildsignal_topdown001 PRIVATE -Wall -Wextra -Wpedantic)
endif()
