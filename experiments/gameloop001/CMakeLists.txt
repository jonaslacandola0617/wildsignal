cmake_minimum_required(VERSION 3.24)
project(WILDSIGNAL_GAMELOOP001 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(CTest)
include(FetchContent)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_subdirectory(../creature3d003 inherited_runtime)
add_library(gameloop_core src/game.cpp src/mesh_clumps.cpp)
target_link_libraries(gameloop_core PUBLIC pixel3d_core)
target_include_directories(gameloop_core PUBLIC src)
if(WIN32)
 target_compile_definitions(gameloop_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0601)
endif()
if(PIXEL3D_BUILD_GL)
 add_executable(wildsignal_gameloop001 src/main.cpp src/gpu.cpp src/hud.cpp src/image.cpp src/mesh_gpu.cpp)
 target_include_directories(wildsignal_gameloop001 BEFORE PRIVATE src ${stb_SOURCE_DIR})
 # FetchContent is scoped by inherited project; obtain the already-populated path.
 FetchContent_GetProperties(stb SOURCE_DIR GAMELOOP_STB)
 target_include_directories(wildsignal_gameloop001 PRIVATE ${GAMELOOP_STB})
 target_link_libraries(wildsignal_gameloop001 PRIVATE gameloop_core SDL3::SDL3-static)
 if(WIN32)
  target_link_libraries(wildsignal_gameloop001 PRIVATE psapi)
 endif()
 execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} OUTPUT_VARIABLE SOURCE_ID OUTPUT_STRIP_TRAILING_WHITESPACE)
 execute_process(COMMAND git status --porcelain WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} OUTPUT_VARIABLE DIRTY OUTPUT_STRIP_TRAILING_WHITESPACE)
 if(DIRTY)
  string(APPEND SOURCE_ID "-dirty")
 endif()
 target_compile_definitions(wildsignal_gameloop001 PRIVATE PIXEL3D_SOURCE_ID="${SOURCE_ID}")
 install(TARGETS wildsignal_gameloop001 RUNTIME DESTINATION .)
 install(DIRECTORY launchers/ DESTINATION .)
 install(FILES README.md DESTINATION .)
endif()
set(GAMELOOP_ASSETS "" CACHE PATH "Private validated GAMELOOP runtime directory")
if(BUILD_TESTING)
 add_executable(gameloop_tests tests/game.cpp)
 target_link_libraries(gameloop_tests PRIVATE gameloop_core)
 add_test(NAME gameloop_state_collision_follower_save COMMAND gameloop_tests)
endif()
if(TARGET wildsignal_gameloop001)
 target_compile_definitions(wildsignal_gameloop001 PRIVATE PETALYN_RUNTIME=1)
endif()
if(BUILD_TESTING AND GAMELOOP_ASSETS)
 add_test(NAME gameloop_actual_private_mesh_skin_lod COMMAND gameloop_tests ${GAMELOOP_ASSETS})
endif()
if(TARGET wildsignal_gameloop001)
 install(FILES tools/hardware.ps1 DESTINATION tools)
endif()
if(BUILD_TESTING AND TARGET wildsignal_gameloop001)
 add_test(NAME gameloop_missing_assets_fail_closed COMMAND wildsignal_gameloop001 --assets ${CMAKE_CURRENT_BINARY_DIR}/absent-private-art --frames 1 --output ${CMAKE_CURRENT_BINARY_DIR}/missing-asset-check)
 set_tests_properties(gameloop_missing_assets_fail_closed PROPERTIES WILL_FAIL TRUE)
 if(GAMELOOP_ASSETS AND PIXEL3D_TEST_GL)
  add_test(NAME gameloop_actual_SDL_loop_reopen_recovery COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/runtime.py $<TARGET_FILE:wildsignal_gameloop001> ${GAMELOOP_ASSETS})
  set_tests_properties(gameloop_actual_SDL_loop_reopen_recovery PROPERTIES TIMEOUT 360)
 endif()
endif()
if(TARGET wildsignal_gameloop001)
 FetchContent_GetProperties(sdl3 SOURCE_DIR GAMELOOP_SDL)
 FetchContent_GetProperties(cgltf SOURCE_DIR GAMELOOP_CGLTF)
 install(FILES ${GAMELOOP_SDL}/LICENSE.txt DESTINATION licenses RENAME SDL3.txt)
 install(FILES ${GAMELOOP_CGLTF}/LICENSE DESTINATION licenses RENAME cgltf.txt)
 install(FILES ${GAMELOOP_STB}/LICENSE DESTINATION licenses RENAME stb.txt)
endif()
