cmake_minimum_required(VERSION 3.24)
project(WILDSIGNAL_2P5D001 LANGUAGES CXX)
option(HYBRID_BUILD_SDL "Build SDL3 hybrid executable" ON)
option(HYBRID_SANITIZE "Use address and undefined behavior sanitizers" OFF)
# Reuse TOPDOWN-001 as a library. No previous prototype source is edited.
set(TOPDOWN001_BUILD_SDL OFF CACHE BOOL "" FORCE)
set(TOPDOWN001_BUILD_LEGACY OFF CACHE BOOL "" FORCE)
set(TOPDOWN001_SANITIZE ${HYBRID_SANITIZE} CACHE BOOL "" FORCE)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_subdirectory(../topdown001 topdown001)
add_library(hybrid_core src/session.cpp src/render.cpp)
target_include_directories(hybrid_core PUBLIC src)
target_link_libraries(hybrid_core PUBLIC topdown001_core)
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} OUTPUT_VARIABLE HYBRID_ID OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
execute_process(COMMAND git status --porcelain WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} OUTPUT_VARIABLE HYBRID_DIRTY OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NOT HYBRID_ID)
 set(HYBRID_ID "UNKNOWN_SOURCE")
endif()
if(HYBRID_DIRTY)
 string(APPEND HYBRID_ID "-dirty")
endif()
target_compile_definitions(hybrid_core PUBLIC HYBRID_BUILD_ID="${HYBRID_ID}")
add_executable(wildsignal_hybrid001_headless src/main.cpp)
target_link_libraries(wildsignal_hybrid001_headless PRIVATE hybrid_core)
if(WIN32)
 target_link_libraries(wildsignal_hybrid001_headless PRIVATE psapi)
endif()
if(HYBRID_BUILD_SDL)
 include(FetchContent)
 set(SDL_SHARED OFF CACHE BOOL "" FORCE)
 set(SDL_STATIC ON CACHE BOOL "" FORCE)
 set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
 set(SDL_TESTS OFF CACHE BOOL "" FORCE)
 set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
 set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
 foreach(subsystem AUDIO CAMERA GPU HAPTIC POWER SENSOR DIALOG)
  set(SDL_${subsystem} OFF CACHE BOOL "" FORCE)
 endforeach()
 set(SDL_RENDER_D3D12 OFF CACHE BOOL "" FORCE)
 set(SDL_RENDER_GPU OFF CACHE BOOL "" FORCE)
 set(SDL_VULKAN OFF CACHE BOOL "" FORCE)
 FetchContent_Declare(SDL3 GIT_REPOSITORY https://github.com/libsdl-org/SDL.git GIT_TAG 877399b2b2cf21e67554ed9046410f268ce1d1b2 GIT_SHALLOW FALSE)
 FetchContent_MakeAvailable(SDL3)
 add_executable(wildsignal_hybrid001 src/main.cpp)
 target_compile_definitions(wildsignal_hybrid001 PRIVATE HYBRID_SDL=1)
 target_link_libraries(wildsignal_hybrid001 PRIVATE hybrid_core SDL3::SDL3-static)
 if(WIN32)
  target_link_libraries(wildsignal_hybrid001 PRIVATE psapi)
 endif()
 install(TARGETS wildsignal_hybrid001 wildsignal_hybrid001_headless RUNTIME DESTINATION .)
 install(FILES ${sdl3_SOURCE_DIR}/LICENSE.txt DESTINATION licenses RENAME SDL3.txt)
 install(FILES README.md DESTINATION .)
 install(DIRECTORY data/ DESTINATION data)
 install(FILES ../topdown001/tools/import_assets.py DESTINATION tools)
endif()
include(CTest)
if(BUILD_TESTING)
 add_executable(hybrid_tests tests/core.cpp)
 target_link_libraries(hybrid_tests PRIVATE hybrid_core)
 add_test(NAME hybrid_invariants COMMAND hybrid_tests)
 find_package(Python3 COMPONENTS Interpreter REQUIRED)
 add_test(NAME hybrid_synthetic_boot COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:wildsignal_hybrid001_headless> ${CMAKE_CURRENT_SOURCE_DIR})
 if(HYBRID_BUILD_SDL)
  add_test(NAME hybrid_sdl_events COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:wildsignal_hybrid001_headless> ${CMAKE_CURRENT_SOURCE_DIR} --native $<TARGET_FILE:wildsignal_hybrid001>)
 endif()
endif()
