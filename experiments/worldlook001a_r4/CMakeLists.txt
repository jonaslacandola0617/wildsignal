cmake_minimum_required(VERSION 3.24)
project(WILDSIGNAL_WORLDLOOK001A_R4 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(WIN32 AND NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Windows x64 only")
endif()
option(PIXEL3D_BUILD_GL "Build real SDL3/OpenGL renderer" ON)
option(PIXEL3D_TEST_GL "Run actual GL synthetic tests (requires GPU or explicitly labelled Mesa and Pillow)" OFF)
option(PIXEL3D_SANITIZE "Address/undefined behavior checks" OFF)
include(FetchContent)
FetchContent_Declare(cgltf GIT_REPOSITORY https://github.com/jkuhlmann/cgltf.git GIT_TAG 360db1a95480fe102ae9c69b27c5d101167ff5ba)
FetchContent_MakeAvailable(cgltf)
add_library(pixel3d_core ../worldlook001a_r3/src/mesh_clumps.cpp src/pixel_surface.cpp ../worldlook001a/src/grass.cpp ../holland004/src/pacing.cpp ../holland001/src/core.cpp ../holland001/src/assets.cpp ../holland001/src/look.cpp ../holland001/src/atmosphere.cpp ../holland001/src/style.cpp ../holland001/src/pixel_camera.cpp ../../src/engine001/json.cpp)
target_include_directories(pixel3d_core PUBLIC src ../worldlook001a_r3/src ../worldlook001a/src ../holland004/src ../holland003/src ../holland001/src ../../src/engine001 PRIVATE ${cgltf_SOURCE_DIR})
if(MSVC)
 target_compile_options(pixel3d_core PRIVATE /W4 /permissive- /utf-8)
else()
 target_compile_options(pixel3d_core PRIVATE -Wall -Wextra -Wpedantic)
 if(PIXEL3D_SANITIZE)
  target_compile_options(pixel3d_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(pixel3d_core PUBLIC -fsanitize=address,undefined)
 endif()
endif()
add_executable(pixel3d_audit ../holland001/src/audit.cpp)
target_link_libraries(pixel3d_audit PRIVATE pixel3d_core)
if(PIXEL3D_BUILD_GL)
 FetchContent_Declare(stb GIT_REPOSITORY https://github.com/nothings/stb.git GIT_TAG 003501f6d7a8668ce8b6d875e34dd97ad0d5d5b5)
 FetchContent_MakeAvailable(stb)
 set(SDL_SHARED OFF CACHE BOOL "" FORCE)
 set(SDL_STATIC ON CACHE BOOL "" FORCE)
 set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
 set(SDL_TESTS OFF CACHE BOOL "" FORCE)
 set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
 set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
 foreach(subsystem AUDIO CAMERA GPU HAPTIC POWER SENSOR DIALOG RENDER)
  set(SDL_${subsystem} OFF CACHE BOOL "" FORCE)
 endforeach()
 set(SDL_VULKAN OFF CACHE BOOL "" FORCE)
 set(SDL_OPENGL ON CACHE BOOL "" FORCE)
 FetchContent_Declare(SDL3 GIT_REPOSITORY https://github.com/libsdl-org/SDL.git GIT_TAG 877399b2b2cf21e67554ed9046410f268ce1d1b2)
 FetchContent_MakeAvailable(SDL3)
 add_executable(wildsignal_worldlook001a_r4 src/main.cpp ../holland004/src/pacing_wait.cpp src/gpu.cpp src/image.cpp src/mesh_gpu.cpp ../worldlook001a/src/grass_gpu.cpp)
 target_include_directories(wildsignal_worldlook001a_r4 PRIVATE ${stb_SOURCE_DIR})
 target_link_libraries(wildsignal_worldlook001a_r4 PRIVATE pixel3d_core SDL3::SDL3-static)
 if(WIN32)
  target_compile_definitions(wildsignal_worldlook001a_r4 PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0601)
  target_link_libraries(wildsignal_worldlook001a_r4 PRIVATE psapi)
 endif()
 execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} OUTPUT_VARIABLE SOURCE_ID OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
 execute_process(COMMAND git status --porcelain WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} OUTPUT_VARIABLE DIRTY OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
 if(DIRTY)
  string(APPEND SOURCE_ID "-dirty")
 endif()
 target_compile_definitions(wildsignal_worldlook001a_r4 PRIVATE PIXEL3D_SOURCE_ID="${SOURCE_ID}")
endif()
include(CTest)
if(BUILD_TESTING)
 add_executable(r4_contract_tests ../worldlook001a_r3/tests/contract.cpp)
 target_link_libraries(r4_contract_tests PRIVATE pixel3d_core)
 add_test(NAME r4_inherited_patch_contract COMMAND r4_contract_tests)
endif()
if(BUILD_TESTING)
 add_executable(r4_surface_tests tests/surface.cpp)
 target_link_libraries(r4_surface_tests PRIVATE pixel3d_core)
 add_test(NAME r4_external_surface_contract COMMAND r4_surface_tests)
endif()
