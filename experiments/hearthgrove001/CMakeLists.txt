cmake_minimum_required(VERSION 3.24)
project(WILDSIGNAL_HEARTHGROVE001 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(CTest)
include(FetchContent)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_subdirectory(../creature3d003 inherited_runtime)
add_library(hearthgrove_core src/game.cpp src/followers.cpp src/mesh_clumps.cpp)
target_link_libraries(hearthgrove_core PUBLIC pixel3d_core)
target_include_directories(hearthgrove_core PUBLIC src)
if(WIN32)
 target_compile_definitions(hearthgrove_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0601)
endif()
if(PIXEL3D_BUILD_GL)
 add_executable(wildsignal_hearthgrove001 src/main.cpp src/gpu.cpp src/hud.cpp src/image.cpp src/mesh_gpu.cpp)
 target_include_directories(wildsignal_hearthgrove001 BEFORE PRIVATE src ${stb_SOURCE_DIR})
 # FetchContent is scoped by inherited project; obtain the already-populated path.
 FetchContent_GetProperties(stb SOURCE_DIR GAMELOOP_STB)
 target_include_directories(wildsignal_hearthgrove001 PRIVATE ${GAMELOOP_STB})
 target_link_libraries(wildsignal_hearthgrove001 PRIVATE hearthgrove_core SDL3::SDL3-static)
 if(WIN32)
  target_link_libraries(wildsignal_hearthgrove001 PRIVATE psapi)
 endif()
 execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} OUTPUT_VARIABLE SOURCE_ID OUTPUT_STRIP_TRAILING_WHITESPACE)
 execute_process(COMMAND git status --porcelain WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} OUTPUT_VARIABLE DIRTY OUTPUT_STRIP_TRAILING_WHITESPACE)
 if(DIRTY)
  string(APPEND SOURCE_ID "-dirty")
 endif()
 target_compile_definitions(wildsignal_hearthgrove001 PRIVATE PIXEL3D_SOURCE_ID="${SOURCE_ID}")
 install(TARGETS wildsignal_hearthgrove001 RUNTIME DESTINATION .)
 install(DIRECTORY launchers/ DESTINATION .)
 install(FILES README.md README_OWNER_REVIEW.md DESTINATION .)
endif()
set(GAMELOOP_ASSETS "" CACHE PATH "Private validated GAMELOOP runtime directory")
if(BUILD_TESTING)
 add_executable(vegetation_profile_tests tests/vegetation.cpp)
 target_link_libraries(vegetation_profile_tests PRIVATE hearthgrove_core)
 add_test(NAME worldart003r2_profiles_culling_lod_validation COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/vegetation.py $<TARGET_FILE:vegetation_profile_tests>)
 add_executable(follower_route_tests tests/followers.cpp)
 target_link_libraries(follower_route_tests PRIVATE hearthgrove_core)
 add_test(NAME gameloop002_trail_route_wait_spacing COMMAND follower_route_tests)
 add_executable(gameloop2_tests tests/game.cpp)
 target_link_libraries(gameloop2_tests PRIVATE hearthgrove_core)
 add_test(NAME gameloop_state_collision_follower_save COMMAND gameloop2_tests)
endif()
if(TARGET wildsignal_hearthgrove001)
 target_compile_definitions(wildsignal_hearthgrove001 PRIVATE PETALYN_RUNTIME=1)
endif()
if(BUILD_TESTING AND GAMELOOP_ASSETS)
 add_test(NAME gameloop_actual_private_mesh_skin_lod COMMAND gameloop2_tests ${GAMELOOP_ASSETS})
endif()
if(TARGET wildsignal_hearthgrove001)
 install(FILES tools/hardware.ps1 DESTINATION tools)
endif()
if(BUILD_TESTING AND TARGET wildsignal_hearthgrove001)
 add_test(NAME gameloop_missing_assets_fail_closed COMMAND wildsignal_hearthgrove001 --assets ${CMAKE_CURRENT_BINARY_DIR}/absent-private-art --frames 1 --output ${CMAKE_CURRENT_BINARY_DIR}/missing-asset-check)
 set_tests_properties(gameloop_missing_assets_fail_closed PROPERTIES WILL_FAIL TRUE)
 if(GAMELOOP_ASSETS AND PIXEL3D_TEST_GL)
  add_test(NAME gameloop_actual_SDL_loop_reopen_recovery COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/runtime.py $<TARGET_FILE:wildsignal_hearthgrove001> ${GAMELOOP_ASSETS})
  set_tests_properties(gameloop_actual_SDL_loop_reopen_recovery PROPERTIES TIMEOUT 360)
 endif()
endif()
if(TARGET wildsignal_hearthgrove001)
 FetchContent_GetProperties(sdl3 SOURCE_DIR GAMELOOP_SDL)
 FetchContent_GetProperties(cgltf SOURCE_DIR GAMELOOP_CGLTF)
 install(FILES ${GAMELOOP_SDL}/LICENSE.txt DESTINATION licenses RENAME SDL3.txt)
 install(FILES ${GAMELOOP_CGLTF}/LICENSE DESTINATION licenses RENAME cgltf.txt)
 install(FILES ${GAMELOOP_STB}/LICENSE DESTINATION licenses RENAME stb.txt)
endif()

 add_executable(hearthgrove_world_tests tests/world.cpp)
 target_link_libraries(hearthgrove_world_tests PRIVATE hearthgrove_core)
 add_test(NAME hearthgrove_world_height_bounds_save_navigation COMMAND hearthgrove_world_tests)
endif()

 add_executable(pixelstyle_policy_tests tests/pixelstyle.cpp)
