cmake_minimum_required(VERSION 3.24)
project(WILDSIGNAL_HEARTHGROVE_EMACEART LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(CTest)
include(FetchContent)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_subdirectory(../creature3d003 inherited_runtime)
add_library(emacegrove_core src/game.cpp src/followers.cpp src/mesh_clumps.cpp)
target_link_libraries(emacegrove_core PUBLIC pixel3d_core)
target_include_directories(emacegrove_core PUBLIC src)
if(WIN32)
 target_compile_definitions(emacegrove_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0601)
endif()
if(PIXEL3D_BUILD_GL)
 add_executable(wildsignal_hearthgrove_emaceart src/main.cpp src/gpu.cpp src/hud.cpp src/image.cpp src/mesh_gpu.cpp)
 target_include_directories(wildsignal_hearthgrove_emaceart BEFORE PRIVATE src ${stb_SOURCE_DIR})
 # FetchContent is scoped by inherited project; obtain the already-populated path.
 FetchContent_GetProperties(stb SOURCE_DIR GAMELOOP_STB)
 target_include_directories(wildsignal_hearthgrove_emaceart PRIVATE ${GAMELOOP_STB})
 target_link_libraries(wildsignal_hearthgrove_emaceart PRIVATE emacegrove_core SDL3::SDL3-static)
 if(WIN32)
  target_link_libraries(wildsignal_hearthgrove_emaceart PRIVATE psapi)
 endif()
 execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} OUTPUT_VARIABLE SOURCE_ID OUTPUT_STRIP_TRAILING_WHITESPACE)
 execute_process(COMMAND git status --porcelain WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} OUTPUT_VARIABLE DIRTY OUTPUT_STRIP_TRAILING_WHITESPACE)
 if(DIRTY)
  string(APPEND SOURCE_ID "-dirty")
 endif()
 target_compile_definitions(wildsignal_hearthgrove_emaceart PRIVATE PIXEL3D_SOURCE_ID="${SOURCE_ID}")
 install(TARGETS wildsignal_hearthgrove_emaceart RUNTIME DESTINATION .)
 # Owner review has ONE playable entrypoint. Retain historical comparison
 # launchers in source, without installing a matrix of runnable variants.
 install(FILES launchers/PLAY_GAME.cmd DESTINATION .)
 install(FILES README.md README_OWNER_REVIEW.md README_SHAPED_RAYS.md README_TREE_OPTIONS.md DESTINATION .)
endif()
